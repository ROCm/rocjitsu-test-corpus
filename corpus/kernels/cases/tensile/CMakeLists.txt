if(NOT "gfx1250" IN_LIST CMAKE_HIP_ARCHITECTURES)
    message(STATUS "Skipping packaged Tensile SGEMM runners: requires gfx1250")
    return()
endif()

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/artifacts.json" artifact_metadata)
string(JSON TENSILE_SOURCE_REVISION GET "${artifact_metadata}" source_revision)

if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "Tensile host code requires the ROCm SDK amdclang++ C++ compiler")
endif()

include(FetchContent)
# Match the revision recorded with the packaged code objects, not a moving branch.
FetchContent_Declare(tensile_rocm_libraries
    URL https://github.com/ROCm/rocm-libraries/archive/${TENSILE_SOURCE_REVISION}.tar.gz
    URL_HASH SHA256=50acc913e6a28f18e3dffde3e2352e8ebd3ba161f2160ca9f762c63601b29ec1
    SOURCE_SUBDIR unused-corpus-host-only)
FetchContent_MakeAvailable(tensile_rocm_libraries)
set(GPU_TARGETS gfx1250)
find_package(hip REQUIRED CONFIG)
find_package(LLVM REQUIRED CONFIG PATHS "${ROCM_PATH}/lib/llvm/lib/cmake/llvm" NO_DEFAULT_PATH)
set(ORIGAMI_ENABLE_INSTALL OFF)
set(ORIGAMI_ENABLE_PYTHON OFF)
set(ORIGAMI_BUILD_TESTING OFF)
set(BUILD_TESTING OFF)
add_subdirectory("${tensile_rocm_libraries_SOURCE_DIR}/shared/origami" origami EXCLUDE_FROM_ALL)
set(TENSILELITE_ENABLE_HOST ON)
set(TENSILELITE_ENABLE_AUTOBUILD OFF)
set(TENSILELITE_ENABLE_CLIENT OFF)
set(TENSILELITE_BUILD_TESTING OFF)
set(HIPBLASLT_ENABLE_YAML ON)
set(HIPBLASLT_BUNDLE_PYTHON_DEPS OFF)
add_subdirectory("${tensile_rocm_libraries_SOURCE_DIR}/projects/hipblaslt/tensilelite" tensilelite EXCLUDE_FROM_ALL)
# HIP numeric headers require HIP language mode even for this host-only adapter.
target_compile_options(tensilelite-host PUBLIC "SHELL:-x hip" --offload-host-only)

string(JSON artifact_directory GET "${artifact_metadata}" artifact_directory)
get_filename_component(TENSILE_ARTIFACT_DIR "${PROJECT_SOURCE_DIR}/../../${artifact_directory}" ABSOLUTE)
# Fail configuration if source-controlled artifacts do not match the pinned set.
foreach(filename TensileLibrary.yaml Kernels.so-000-gfx1250.hsaco TensileLibrary_gfx1250.co)
    string(JSON expected_hash GET "${artifact_metadata}" files "${filename}")
    file(SHA256 "${TENSILE_ARTIFACT_DIR}/${filename}" actual_hash)
    if(NOT actual_hash STREQUAL expected_hash)
        message(FATAL_ERROR "Tensile artifact hash mismatch: ${filename}")
    endif()
endforeach()
set(TENSILE_ARTIFACT_HASHES "")
foreach(filename TensileLibrary.yaml Kernels.so-000-gfx1250.hsaco TensileLibrary_gfx1250.co)
    string(JSON expected_hash GET "${artifact_metadata}" files "${filename}")
    string(APPEND TENSILE_ARTIFACT_HASHES "    {\"${filename}\", \"${expected_hash}\"},\n")
endforeach()
set(TENSILE_KERNEL_NAMES "")
foreach(index RANGE 0 1)
    string(JSON kernel_name GET "${artifact_metadata}" solutions ${index} kernel_name)
    string(APPEND TENSILE_KERNEL_NAMES "    \"${kernel_name}\",\n")
endforeach()
configure_file(artifacts.hpp.in artifacts.hpp @ONLY)
configure_file(artifacts.json provenance.json COPYONLY)
foreach(index RANGE 0 1)
    string(JSON layout GET "${artifact_metadata}" solutions ${index} layout)
    set(case_name "tensile_sgemm_gfx1250_mt${layout}")
    add_executable(${case_name} runner.cpp)
    target_compile_features(${case_name} PRIVATE cxx_std_17)
    target_compile_definitions(${case_name} PRIVATE TENSILE_SOLUTION_INDEX=${index})
    target_include_directories(${case_name} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${PROJECT_SOURCE_DIR}/include" ${LLVM_INCLUDE_DIRS})
    target_link_libraries(${case_name} PRIVATE tensilelite::tensilelite-host hip::host LLVMObjectYAML)
endforeach()

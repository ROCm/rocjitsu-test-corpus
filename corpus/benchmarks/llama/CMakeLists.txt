cmake_minimum_required(VERSION 3.28)
project(rocjitsu_bounded_ggml LANGUAGES C CXX)
if(NOT IS_ABSOLUTE "${LLAMA_SOURCE_DIR}" OR NOT EXISTS "${LLAMA_SOURCE_DIR}/ggml/CMakeLists.txt")
  message(FATAL_ERROR "Pass the verified absolute LLAMA_SOURCE_DIR via prepare.py")
endif()
set(GGML_CCACHE OFF CACHE BOOL "No shared compiler cache" FORCE)
set(GGML_CUDA_FA OFF CACHE BOOL "No FlashAttention" FORCE)
set(GGML_HIP_GRAPHS OFF CACHE BOOL "One graph invocation per sample" FORCE)
set(GGML_HIP_NO_VMM ON CACHE BOOL "No VMM" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "Portable CPU build" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "No runtime backend discovery" FORCE)
if(GGML_HIP)
  set(GPU_TARGETS "${CMAKE_HIP_ARCHITECTURES}" CACHE STRING "Explicit HIP targets from preparation" FORCE)
endif()
add_subdirectory("${LLAMA_SOURCE_DIR}/ggml" "${CMAKE_CURRENT_BINARY_DIR}/ggml")
add_executable(bounded-ops bounded-ops.cpp)
target_compile_features(bounded-ops PRIVATE cxx_std_17)
target_link_libraries(bounded-ops PRIVATE ggml)
if(GGML_HIP)
  find_package(hip REQUIRED)
  target_link_libraries(bounded-ops PRIVATE hip::host)
  target_compile_definitions(bounded-ops PRIVATE EXPERIMENT_HIP)
endif()
option(BENCHMARK_BUILD_CPU_MUTATION "Build host-only correctness negative controls" OFF)
if(BENCHMARK_BUILD_CPU_MUTATION)
  if(GGML_HIP)
    message(FATAL_ERROR "Mutation is CPU-only")
  endif()
  add_executable(bounded-ops-noop bounded-ops.cpp)
  target_compile_features(bounded-ops-noop PRIVATE cxx_std_17)
  target_compile_definitions(bounded-ops-noop PRIVATE BENCHMARK_CPU_NOOP)
  target_link_libraries(bounded-ops-noop PRIVATE ggml)
  add_executable(bounded-ops-partial bounded-ops.cpp)
  target_compile_features(bounded-ops-partial PRIVATE cxx_std_17)
  target_compile_definitions(bounded-ops-partial PRIVATE BENCHMARK_CPU_PARTIAL)
  target_link_libraries(bounded-ops-partial PRIVATE ggml)
  add_executable(bounded-ops-corrupt bounded-ops.cpp)
  target_compile_features(bounded-ops-corrupt PRIVATE cxx_std_17)
  target_compile_definitions(bounded-ops-corrupt PRIVATE BENCHMARK_CPU_CORRUPT)
  target_link_libraries(bounded-ops-corrupt PRIVATE ggml)
  add_executable(bounded-ops-wrong-row bounded-ops.cpp)
  target_compile_features(bounded-ops-wrong-row PRIVATE cxx_std_17)
  target_compile_definitions(bounded-ops-wrong-row PRIVATE BENCHMARK_CPU_WRONG_ROW)
  target_link_libraries(bounded-ops-wrong-row PRIVATE ggml)
endif()

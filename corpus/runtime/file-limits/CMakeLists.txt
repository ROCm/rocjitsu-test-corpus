cmake_minimum_required(VERSION 3.28)
if(NOT CMAKE_HIP_ARCHITECTURES)
    message(FATAL_ERROR "File-limit tests require -DCMAKE_HIP_ARCHITECTURES=<gfx target> matching ROCJITSU_CONFIG")
endif()
list(LENGTH CMAKE_HIP_ARCHITECTURES architecture_count)
if(NOT architecture_count EQUAL 1)
    message(FATAL_ERROR "File-limit tests require exactly one HIP architecture matching ROCJITSU_CONFIG")
endif()
set(ROCM_PATH "$ENV{ROCM_PATH}" CACHE PATH "ROCm SDK root")
if(ROCM_PATH)
    list(APPEND CMAKE_PREFIX_PATH "${ROCM_PATH}")
    set(CMAKE_HIP_COMPILER_ROCM_ROOT "${ROCM_PATH}" CACHE PATH "HIP compiler root")
    if(NOT CMAKE_HIP_COMPILER)
        find_program(CMAKE_HIP_COMPILER amdclang++ HINTS "${ROCM_PATH}/lib/llvm/bin" REQUIRED)
    endif()
endif()
project(rocjitsu_file_limit_tests LANGUAGES CXX HIP)
enable_testing()
find_package(hip REQUIRED CONFIG)
find_package(GTest QUIET)
if(NOT TARGET GTest::gtest)
    include(FetchContent)
    set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(googletest
        GIT_REPOSITORY https://github.com/google/googletest.git
        GIT_TAG v1.15.2 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(googletest)
endif()
add_executable(hip_rlimit_fsize_test hip_rlimit_fsize_test.hip)
set_target_properties(hip_rlimit_fsize_test PROPERTIES HIP_STANDARD 20 HIP_STANDARD_REQUIRED ON)
target_link_libraries(hip_rlimit_fsize_test PRIVATE GTest::gtest hip::device)
set(FILE_LIMIT_SANITIZER "" CACHE STRING "Host sanitizer: address,undefined or thread")
if(FILE_LIMIT_SANITIZER)
    target_compile_options(hip_rlimit_fsize_test PRIVATE
        "SHELL:-Xarch_host -fsanitize=${FILE_LIMIT_SANITIZER}"
        "SHELL:-Xarch_host -fno-sanitize-recover=all")
    target_link_options(hip_rlimit_fsize_test PRIVATE
        "SHELL:-Xarch_host -fsanitize=${FILE_LIMIT_SANITIZER}"
        "SHELL:-Xarch_host -shared-libsan")
endif()
set(ROCJITSU "" CACHE FILEPATH "rocjitsu executable")
set(ROCJITSU_CONFIG "" CACHE FILEPATH "Matching simulator KMD configuration")
if(NOT ROCJITSU OR NOT ROCJITSU_CONFIG)
    message(STATUS "Set ROCJITSU and ROCJITSU_CONFIG to register simulator tests")
    return()
endif()
foreach(mode IN ITEMS local daemon)
    set(daemon_arg)
    if(mode STREQUAL "daemon")
        set(daemon_arg --daemon)
    endif()
    foreach(phase IN ITEMS early late inherited fork)
        set(limits zero hard soft)
        if(phase STREQUAL "inherited")
            set(limits hard soft)
        elseif(phase STREQUAL "fork")
            set(limits soft-zero)
        endif()
        foreach(limit IN LISTS limits)
            set(prefix)
            set(expected_limits)
            if(phase STREQUAL "inherited")
                if(limit STREQUAL "hard")
                    set(prefix prlimit --fsize=2097152:2097152)
                    set(expected_limits 2097152 2097152)
                else()
                    set(prefix prlimit --fsize=2097152:unlimited)
                    set(expected_limits 2097152 unlimited)
                endif()
            endif()
            set(symbolizer_env)
            if(FILE_LIMIT_SANITIZER STREQUAL "thread" AND phase STREQUAL "fork")
                # The sanitizer snapshots its environment before main. A child
                # symbolizer inherits LD_PRELOAD and FSIZE=0; prevent its optional
                # temporary-file shadow initialization before any runtime starts.
                set(symbolizer_env "TMPDIR=/dev/null")
            endif()
            set(name HipFileSizeLimitTest.${mode}.${phase}.${limit})
            add_test(NAME ${name} COMMAND ${prefix} "${ROCJITSU}" ${daemon_arg}
                --config "${ROCJITSU_CONFIG}" -- $<TARGET_FILE:hip_rlimit_fsize_test>
                ${mode} ${phase} ${limit} ${expected_limits})
            set_tests_properties(${name} PROPERTIES TIMEOUT 120
                ENVIRONMENT "ROCJITSU_RUNTIME_DIR=runtime/fsize-${mode}-${phase}-${limit};ASAN_OPTIONS=detect_leaks=0;TSAN_OPTIONS=suppressions=${CMAKE_CURRENT_SOURCE_DIR}/tsan_hip_runtime.supp;${symbolizer_env}")
        endforeach()
    endforeach()
endforeach()

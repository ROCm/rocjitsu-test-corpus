cmake_minimum_required(VERSION 3.20)
project(runtime_cts LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  message(FATAL_ERROR "These KFD tests currently require Linux x86-64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Preserve flat binary paths, including multiprocess's sibling queue_flood lookup.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
find_package(Threads REQUIRED)
find_package(Python3 3.11 COMPONENTS Interpreter REQUIRED)
add_library(cts_common STATIC common/test.cc)
target_include_directories(cts_common PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(cts_common PUBLIC -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
target_link_libraries(cts_common PUBLIC Threads::Threads)
set(AMDGPU_LLVM_BIN "" CACHE PATH "AMDGPU-enabled clang and ld.lld directory")
include(cmake/RuntimeCTS.cmake)
include(cmake/Requirements.cmake)
include(cmake/Platforms.cmake)

set(CTS_ARCHS "gfx1201" CACHE STRING "Kernel/platform targets (semicolon separated)")
set(CTS_SUITES "aql;pm4" CACHE STRING "Suites to build (aql and/or pm4)")
foreach(suite IN LISTS CTS_SUITES)
  if(NOT suite MATCHES "^(aql|pm4)$")
    message(FATAL_ERROR "Unknown suite: ${suite}")
  endif()
endforeach()
if(NOT CTS_ARCHS OR NOT CTS_SUITES)
  message(FATAL_ERROR "Select at least one target and suite")
endif()
cts_requirements()
foreach(arch IN LISTS CTS_ARCHS)
  cts_platform(${arch})
  set(kernel_dir ${CMAKE_BINARY_DIR}/kernels/${arch})
  cts_check_kernel_toolchain(${arch} work)
  foreach(kernel work gate lds pipeline geometry host_exchange payload)
    cts_kernel(${kernel})
  endforeach()
  if("wgp_placement" IN_LIST platform_features)
    cts_kernel(placement)
  endif()
  if("wave32_scratch" IN_LIST platform_features)
    cts_kernel(scratch)
  endif()
  if("aql_metadata" IN_LIST platform_features)
    cts_check_kernel_toolchain(${arch} metadata
      -mllvm -amdgpu-kernarg-preload -mllvm -amdgpu-kernarg-preload-count=29)
    cts_kernel(metadata -mllvm -amdgpu-kernarg-preload -mllvm -amdgpu-kernarg-preload-count=29)
  endif()
  string(REPLACE ";" "\n" features "${platform_features}")
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/runtime-${arch}-features.txt" CONTENT "${features}\n")
  foreach(suite IN LISTS CTS_SUITES)
    add_library(kfd_support_${suite}_${arch} STATIC
      ${PROJECT_SOURCE_DIR}/common/support/kfd.cc ${PROJECT_SOURCE_DIR}/common/support/sdma.cc)
    target_include_directories(kfd_support_${suite}_${arch} PUBLIC ${PROJECT_SOURCE_DIR}/common)
    target_compile_definitions(kfd_support_${suite}_${arch} PUBLIC
      CTS_TARGET_NAME="${arch}" CTS_GFX_VERSION=${gfx_version})
    target_link_libraries(kfd_support_${suite}_${arch} PUBLIC cts_common)
    if(suite STREQUAL "aql")
      target_compile_definitions(kfd_support_${suite}_${arch} PUBLIC CTS_AQL_ONLY=1)
      add_subdirectory(aql aql/${arch})
    else()
      target_include_directories(kfd_support_${suite}_${arch} PUBLIC
        ${PROJECT_SOURCE_DIR}/pm4/support/${pm4_profile})
      add_subdirectory(pm4 pm4/${arch})
    endif()
  endforeach()
endforeach()

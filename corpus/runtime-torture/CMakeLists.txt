cmake_minimum_required(VERSION 3.20)
project(mec_mes_torture LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  message(FATAL_ERROR "These KFD tests currently require Linux x86-64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(kfd_test_support STATIC support/kfd.cc support/sdma.cc)
target_include_directories(kfd_test_support PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(kfd_test_support PUBLIC -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
target_link_libraries(kfd_test_support PUBLIC Threads::Threads)
enable_testing()
set(AMDGPU_LLVM_BIN "" CACHE PATH "AMDGPU-enabled clang and ld.lld directory")
option(TORTURE_INVESTIGATION_TESTS "Register unresolved live-wave pause and AQL oversubscription reproducers" OFF)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
foreach(arch IN ITEMS 1201 1250)
  foreach(test IN ITEMS packet_flood queue_flood queue_churn dependency_chain dependency_diamond
      indirect_buffers concurrent_submit multiprocess atomic_contention random_dag
      queue_pause_resume wait_comparisons timeline_points timeline_fuzz gpu_ping_pong
      host_gate private_dma allocation_churn shared_ib sdma_flood engine_pipeline concurrent_queue_churn)
    add_executable(${test}_gfx${arch} ${test}.cc)
    target_compile_definitions(${test}_gfx${arch} PRIVATE TEST_GFX=${arch})
    target_link_libraries(${test}_gfx${arch} PRIVATE kfd_test_support)
    add_test(NAME ${test}_gfx${arch} COMMAND ${test}_gfx${arch})
    set_tests_properties(${test}_gfx${arch} PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77 RUN_SERIAL TRUE LABELS "smoke")
  endforeach()
  foreach(kind IN ITEMS and or)
    set(test aql_barrier_${kind}_gfx${arch})
    add_executable(${test} aql_barriers.cc)
    target_compile_definitions(${test} PRIVATE TEST_GFX=${arch} AQL_WAIT_ANY=$<STREQUAL:${kind},or>)
    target_link_libraries(${test} PRIVATE kfd_test_support)
    add_test(NAME ${test} COMMAND ${test})
    set_tests_properties(${test} PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77 RUN_SERIAL TRUE LABELS "smoke")
  endforeach()
endforeach()

if(AMDGPU_LLVM_BIN)
  foreach(arch IN ITEMS 1201 1250)
    set(kernel_dir ${CMAKE_CURRENT_BINARY_DIR}/gfx${arch})
    file(MAKE_DIRECTORY ${kernel_dir})
    foreach(kernel IN ITEMS work gate lds)
      add_custom_command(OUTPUT ${kernel_dir}/${kernel}_kernel.inc
        COMMAND ${AMDGPU_LLVM_BIN}/clang -x cl --target=amdgcn-amd-amdhsa
          -mcpu=gfx${arch} -mno-wavefrontsize64 -nogpulib -O2 -c
          ${CMAKE_CURRENT_SOURCE_DIR}/kernels/${kernel}.cl -o ${kernel_dir}/${kernel}.o
        COMMAND ${AMDGPU_LLVM_BIN}/ld.lld -shared --build-id=none
          ${kernel_dir}/${kernel}.o -o ${kernel_dir}/${kernel}.co
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/support/embed_kernel.py
          ${kernel_dir}/${kernel}.co ${kernel_dir}/${kernel}_kernel.inc
        DEPENDS kernels/${kernel}.cl support/embed_kernel.py VERBATIM)
    endforeach()
    foreach(test IN ITEMS dispatch_flood dispatch_barrier running_queue_pause lds_dispatch)
      if(test STREQUAL "dispatch_flood")
        set(kernel work)
      elseif(test STREQUAL "lds_dispatch")
        set(kernel lds)
      else()
        set(kernel gate)
      endif()
      add_executable(${test}_gfx${arch} ${test}.cc ${kernel_dir}/${kernel}_kernel.inc)
      target_include_directories(${test}_gfx${arch} PRIVATE ${kernel_dir})
      target_compile_definitions(${test}_gfx${arch} PRIVATE TEST_GFX=${arch})
      target_link_libraries(${test}_gfx${arch} PRIVATE kfd_test_support)
      if(NOT test STREQUAL "running_queue_pause" OR TORTURE_INVESTIGATION_TESTS)
        add_test(NAME ${test}_gfx${arch} COMMAND ${test}_gfx${arch})
        set(label smoke)
        if(test STREQUAL "running_queue_pause")
          set(label investigation)
        endif()
        set_tests_properties(${test}_gfx${arch} PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77 RUN_SERIAL TRUE LABELS ${label})
      endif()
    endforeach()
    if(TORTURE_INVESTIGATION_TESTS)
      add_test(NAME dispatch_oversubscribed_gfx${arch} COMMAND dispatch_flood_gfx${arch} --queues 8 --iterations 1)
      set_tests_properties(dispatch_oversubscribed_gfx${arch} PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77 RUN_SERIAL TRUE LABELS "investigation")
    endif()
  endforeach()
else()
  message(STATUS "Set AMDGPU_LLVM_BIN to build shader-dispatch tests")
endif()

cmake_minimum_required(VERSION 3.20)
project(mec_mes_torture LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  message(FATAL_ERROR "These KFD tests currently require Linux x86-64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Preserve flat binary paths, including multiprocess's sibling queue_flood lookup.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
find_package(Threads REQUIRED)
find_package(Python3 3.11 COMPONENTS Interpreter REQUIRED)
add_library(torture_common STATIC common/test.cc)
target_include_directories(torture_common PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(torture_common PUBLIC -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
target_link_libraries(torture_common PUBLIC Threads::Threads)
set(AMDGPU_LLVM_BIN "" CACHE PATH "AMDGPU-enabled clang and ld.lld directory")
include(cmake/RuntimeTorture.cmake)
include(cmake/Requirements.cmake)

set(TORTURE_ARCHS "gfx1201" CACHE STRING "Architecture directories to build (semicolon separated)")
set(unsupported_archs ${TORTURE_ARCHS})
list(REMOVE_ITEM unsupported_archs gfx1201 gfx1250)
if(NOT "${unsupported_archs}" STREQUAL "" OR "${TORTURE_ARCHS}" STREQUAL "")
  message(FATAL_ERROR "TORTURE_ARCHS must select gfx1201 and/or gfx1250")
endif()

torture_requirements()
if("gfx1201" IN_LIST TORTURE_ARCHS)
  torture_check_kernel_toolchain(gfx1201 work)
endif()
if("gfx1250" IN_LIST TORTURE_ARCHS)
  torture_check_kernel_toolchain(gfx1250 metadata
    -mllvm -amdgpu-kernarg-preload -mllvm -amdgpu-kernarg-preload-count=29)
endif()

if("gfx1201" IN_LIST TORTURE_ARCHS)
  add_subdirectory(gfx1201)
endif()
if("gfx1250" IN_LIST TORTURE_ARCHS)
  add_subdirectory(gfx1250)
endif()
